```sql

create table if not EXISTS table1 (
id int,
age string,
name string,
score string,
sex string
)USING paimon
tblproperties (
'primary-key' = 'id'

(1, '11','林大','100','男'),
(2, '12','王二','99','男');
(3, '13','张三','99','男');
(4, '14','李四','100','女');
SELECT * FROM table1;
drop table if exists table2;
create table if not EXISTS table2 (
id int,
age string,
name string,
score string,
sex string
)USING paimon
tblproperties (
'primary-key' = 'id'
