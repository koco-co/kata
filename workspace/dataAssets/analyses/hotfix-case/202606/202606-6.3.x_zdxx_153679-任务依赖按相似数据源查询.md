```

DROP TABLE IF EXISTS ads_ct_lifecycle_kpi;

CREATE TABLE ads_ct_lifecycle_kpi (
  kpi_id VARCHAR(64) COMMENT '指标ID',
  stat_date DATE COMMENT '统计日期',
  lifecycle_stage VARCHAR(64) COMMENT '生命周期阶段',
  kpi_value DECIMAL(18,2) COMMENT '指标值'
)
COMMENT 'hotfix_153679_任务依赖相似数据源回归表'
DUPLICATE KEY(kpi_id, stat_date)
DISTRIBUTED BY HASH(kpi_id) BUCKETS 1
PROPERTIES ("replication_num" = "1");

INSERT INTO ads_ct_lifecycle_kpi VALUES
('KPI_001', '2026-06-25', 'active', 100.00),
('KPI_002', '2026-06-25', 'silent', 20.00);

SELECT COUNT(1) AS cnt FROM ads_ct_lifecycle_kpi;
-- 预期结果：2
6. 本次回归必须准备或定位禅道同类任务链路：
   - 上游任务名称：${UpstreamDorisSqlTask}
   - 上游任务类型：DorisSQL
   - 上游任务脚本：引用 ${SchemaA}.ads_ct_lifecycle_kpi
   - 下游任务名称：${DownstreamDataSyncTask}
   - 下游任务类型：数据同步
   - 下游任务关系：与 ${UpstreamDorisSqlTask} 形成上下游链路，并与 ads_ct_lifecycle_kpi 形成任务血缘
```
